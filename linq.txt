using System;
using System.Collections.Generic;
using System.Linq;
using System.Web;
using System.Web.UI;
using System.Web.UI.WebControls;

namespace WebApplication5
{
    public partial class WebForm1 : System.Web.UI.Page
    {
        [Serializable]

        public class student
        {
            public string name {  get; set; }
            public int age { get; set; }
            public int password { get; set; }
            public string course { get; set; }
        }

        private List<student> stud
        {
            get
            {
                if (ViewState["stud"] == null)
                    ViewState["stud"] = new List<student>();
                return (List<student>)ViewState["stud"];
            }
        }
        protected void Page_Load(object sender, EventArgs e)
        {
         
        }

        protected void btnadd_Click(object sender, EventArgs e)
        {
            string name1 = txtname.Text;
            int age1 = int.Parse(txtage.Text);
            int password1 = int.Parse(txtpassword.Text);
            string course1 = txtcourse.Text;

            stud.Add(new student { name = name1, age = age1, password = password1, course = course1 });

            Label5.Text = "Add succesfully";

        }

        protected void btndisplay_Click(object sender, EventArgs e)
        {
            foreach(var s in stud)
            {
                Label5.Text += s.name.ToString() + "<br/>" + 
                    s.age + "<br/>" + 
                    s.password + "<br/>" + 
                    s.course ;
            }
        }

        protected void btnsearch_Click(object sender, EventArgs e)
        {
            var name = stud
                .Where(s => s.name.StartsWith("r", StringComparison.OrdinalIgnoreCase))
                .Select(s => s.name);

            Label5.Text = "This Name Start Withs M like " + string.Join(",",name);
        }

        protected void btncount_Click(object sender, EventArgs e)
        {
            int count = stud.Count();

            Label5.Text = count.ToString();
        }

        protected void btnsum_Click(object sender, EventArgs e)
        {
            int sum = stud.Sum(s => s.age);
            Label5.Text = sum.ToString();
        }

        protected void btnfirst_Click(object sender, EventArgs e)
        {
            var first = stud.First().name;
            Label5.Text = first.ToString();
        }

        protected void btnlast_Click(object sender, EventArgs e)
        {
            var last = stud.Last().name;
            Label5.Text = last.ToString();
        }

        protected void btnmin_Click(object sender, EventArgs e)
        {
            int min = stud.Min(s => s.age);
            Label5.Text = min.ToString();
        }
    }
}
