using MySql.Data.MySqlClient;
using System;
using System.Collections.Generic;
using System.Configuration;
using System.Data;
using System.Linq;
using System.Web;
using System.Web.UI;
using System.Web.UI.WebControls;

namespace WebApplication2
{
    public partial class WebForm1 : System.Web.UI.Page
    {
      
        private void loaddata()
        {
            string connstr = ConfigurationManager.ConnectionStrings["MySqlConn"].ConnectionString;

            using(MySqlConnection conn = new MySqlConnection(connstr))
            {
                string query = "SELECT * FROM student";
                MySqlDataAdapter ap = new MySqlDataAdapter(query, conn);
                DataTable dt = new DataTable();
                ap.Fill(dt);
                GridView1.DataSource = dt;
                GridView1.DataBind();
            }
        }
        protected void Page_Load(object sender, EventArgs e)
        {
            if (!IsPostBack)
            {
                loaddata();
            }
        }

        protected void btninsert_Click(object sender, EventArgs e)
        {
            int roll = int.Parse(txtroll.Text);
            string name = txtname.Text;
            string course = txtcourse.Text;
            string email = txtemail.Text;

            string connstr = ConfigurationManager.ConnectionStrings["MySqlConn"].ConnectionString;

            using (MySqlConnection conn = new MySqlConnection(connstr))
            {
                string query = "INSERT INTO student(roll,name,course,email) VALUES(@roll,@name,@course,@email)";
                MySqlCommand cmd = new MySqlCommand(query, conn);
                cmd.Parameters.AddWithValue("@roll", roll);
                cmd.Parameters.AddWithValue("@name", name);
                cmd.Parameters.AddWithValue("@course", course);
                cmd.Parameters.AddWithValue("@email", email);

                conn.Open();
                cmd.ExecuteNonQuery();
                conn.Close();
                loaddata();
                txtroll.Text = "";
                txtname.Text = "";
                txtcourse.Text = "";
                txtemail.Text = "";

            }
        }

        protected void btnupdate_Click(object sender, EventArgs e)
        {
            int roll = int.Parse(txtroll.Text);
            string name = txtname.Text;
            string course = txtcourse.Text;
            string email = txtemail.Text;

            string connstr = ConfigurationManager.ConnectionStrings["MySqlConn"].ConnectionString;

            using (MySqlConnection conn = new MySqlConnection(connstr))
            {
                string query = "UPDATE student SET name=@name, course=@course, email=@email WHERE roll=@roll";
                MySqlCommand cmd = new MySqlCommand(query, conn);
                cmd.Parameters.AddWithValue("@roll", roll);
                cmd.Parameters.AddWithValue("@name", name);
                cmd.Parameters.AddWithValue("@course", course);
                cmd.Parameters.AddWithValue("@email", email);

                conn.Open();
                cmd.ExecuteNonQuery();
                conn.Close();

                loaddata();
                txtroll.Text = "";
                txtname.Text = "";
                txtcourse.Text = "";
                txtemail.Text = "";
            }
        }

        protected void btndelete_Click(object sender, EventArgs e)
        {
            int roll = int.Parse(txtroll.Text);
           

            string connstr = ConfigurationManager.ConnectionStrings["MySqlConn"].ConnectionString;

            using (MySqlConnection conn = new MySqlConnection(connstr))
            {
                string query = "DELETE FROM student WHERE roll = @roll";
                MySqlCommand cmd = new MySqlCommand(query, conn);
                cmd.Parameters.AddWithValue("@roll", roll);
      

                conn.Open();
                cmd.ExecuteNonQuery();
                conn.Close();

                loaddata();
                txtroll.Text = "";
                txtname.Text = "";
                txtcourse.Text = "";
                txtemail.Text = "";
            }
        }
    }
}
